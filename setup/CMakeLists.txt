# mpr_setup: first-run setup for Metroid Prime Remastered PC (dump -> game folder). Windows x64, clang-cl + Ninja.
#   pwsh -File tools\setup\scripts\build.ps1 [-Clean]
cmake_minimum_required(VERSION 3.20)
# Static MSVC runtime (/MT): players need no Visual C++ Redistributable. Set before project() so every target below, third-party code
# included, uses it; the libraries this one links must agree (a mismatch is a link error).
if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
project(mpr_setup C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
enable_testing()

# Libraries from the S4 tools (built in-tree, not installed): ASTC->BC7 transcoder and Tegra tiling.
set(MPRS_TEXTURES_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../textures CACHE PATH "tools/textures")
set(MPRS_TILING_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../tiling CACHE PATH "tools/tiling")
add_subdirectory(${MPRS_TEXTURES_DIR} ${CMAKE_BINARY_DIR}/textures EXCLUDE_FROM_ALL)
add_subdirectory(${MPRS_TILING_DIR} ${CMAKE_BINARY_DIR}/tiling EXCLUDE_FROM_ALL)

set(S ${CMAKE_CURRENT_SOURCE_DIR}/src)
set(MPRS_WARN /W3 -Wno-unused-parameter)

# Adds a static library from the listed sources that exist (so partial trees still configure).
function(mprs_lib name)
    set(srcs "")
    foreach(f ${ARGN})
        if(EXISTS ${S}/${f})
            list(APPEND srcs ${S}/${f})
        endif()
    endforeach()
    if(srcs)
        add_library(${name} STATIC ${srcs})
        target_include_directories(${name} PUBLIC ${S})
        target_compile_definitions(${name} PUBLIC _CRT_SECURE_NO_WARNINGS UNICODE _UNICODE)
        target_compile_options(${name} PRIVATE ${MPRS_WARN})
        set(MPRS_HAVE_${name} ON PARENT_SCOPE)
    endif()
endfunction()

mprs_lib(mprs_base util.cpp storage.cpp log.cpp)
mprs_lib(mprs_crypto crypto.cpp fileio.cpp)
if(TARGET mprs_crypto)
    target_link_libraries(mprs_crypto PUBLIC mprs_base bcrypt)
endif()
mprs_lib(mprs_fs pfs.cpp romfs.cpp)
if(TARGET mprs_fs)
    target_link_libraries(mprs_fs PUBLIC mprs_base mprs_crypto)
endif()
mprs_lib(mprs_pak lzss.cpp pak.cpp txtr.cpp)
if(TARGET mprs_pak)
    target_link_libraries(mprs_pak PUBLIC mprs_base)
endif()
# The rest of the core (keys, NCA, containers, texture cache, shader cache, pipeline) lives in mprs_core.
mprs_lib(mprs_core keys.cpp nca.cpp container.cpp nso.cpp texcache.cpp shadercache.cpp programs.cpp pipeline.cpp ncz.cpp)
if(TARGET mprs_core)
    target_link_libraries(mprs_core PUBLIC mprs_base mprs_crypto mprs_fs mprs_pak tegra_tiling astc2bc7 shlwapi ole32 shell32)
    # miniz 3.1.2 (MIT, third_party/miniz, from the official release): zlib inflate for the shader program list (programs.cpp)
    add_library(miniz_lib STATIC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/miniz/miniz.c)
    target_include_directories(miniz_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/miniz)
    target_compile_definitions(miniz_lib PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(mprs_core PUBLIC miniz_lib)
    add_dependencies(mprs_core astc2bc7_cli)   # builds the three astcenc decoder DLLs next to the executables
    # BC5 encoder (rgbcx) and the reference BC7 decoder (the oracle's): two-channel normal maps and the RGBA8 fallback of texcache.cpp
    set(BC7_TP ${MPRS_TEXTURES_DIR}/third_party/bc7enc_rdo)
    target_sources(mprs_core PRIVATE ${BC7_TP}/rgbcx.cpp ${BC7_TP}/bc7decomp_ref.cpp)
    target_include_directories(mprs_core PRIVATE ${BC7_TP})
endif()

# zstd (BSD-3-Clause; tools/setup/scripts/fetch_zstd.ps1 puts it in third_party/zstd): NCZ/NSZ/XCZ support and shader packs
if(TARGET mprs_core AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/third_party/zstd/lib/zstd.h)
    file(GLOB ZSTD_SRC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/zstd/lib/common/*.c
         ${CMAKE_CURRENT_SOURCE_DIR}/third_party/zstd/lib/compress/*.c ${CMAKE_CURRENT_SOURCE_DIR}/third_party/zstd/lib/decompress/*.c)
    add_library(zstd_lib STATIC ${ZSTD_SRC})
    target_include_directories(zstd_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/zstd/lib)
    target_compile_definitions(zstd_lib PRIVATE ZSTD_DISABLE_ASM=1 _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(mprs_core PUBLIC zstd_lib)
    target_compile_definitions(mprs_core PUBLIC MPRS_HAVE_ZSTD=1)
endif()

# ---- tests: tests/test_<name>.cpp -> test_<name>.exe; libraries listed per test below, everything else gets mprs_core
function(mprs_test name)
    set(src ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${name}.cpp)
    if(NOT EXISTS ${src})
        return()
    endif()
    file(GLOB helpers ${CMAKE_CURRENT_SOURCE_DIR}/tests/synth*.cpp)
    add_executable(test_${name} ${src} ${helpers})
    target_include_directories(test_${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests ${S})
    target_compile_definitions(test_${name} PRIVATE _CRT_SECURE_NO_WARNINGS UNICODE _UNICODE)
    target_compile_options(test_${name} PRIVATE ${MPRS_WARN})
    target_link_libraries(test_${name} PRIVATE ${ARGN})
    add_test(NAME ${name} COMMAND test_${name})
endfunction()

mprs_test(crypto mprs_crypto mprs_base)
mprs_test(pfs mprs_fs mprs_crypto mprs_base)
mprs_test(romfs mprs_fs mprs_crypto mprs_base)
mprs_test(pak mprs_pak mprs_base)
foreach(t keys nca container texcache shadercache programs pipeline nso zstd realdump)
    mprs_test(${t} mprs_core)
endforeach()
